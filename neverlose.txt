local function corner(parent, radius)
    local c = Instance.new("UICorner", parent)
    c.CornerRadius = UDim.new(0, radius or 8)
    return c
end
local function stroke(parent, color, thickness, transparency)
    local s = Instance.new("UIStroke", parent)
    s.Color = color or Theme.Border
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    return s
end
local function padding(parent, top, right, bottom, left)
    local p = Instance.new("UIPadding", parent)
    p.PaddingTop = UDim.new(0, top or 0)
    p.PaddingRight = UDim.new(0, right or top or 0)
    p.PaddingBottom = UDim.new(0, bottom or top or 0)
    p.PaddingLeft = UDim.new(0, left or right or top or 0)
    return p
end
local function tween(obj, time, props, style, dir)
    local ti = TweenInfo.new(time or 0.2, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out)
    local t = TweenService:Create(obj, ti, props)
    t:Play()
    return t
end

    Circle.BackgroundColor3 = Color3.new(1, 1, 1)
    Circle.BorderSizePixel = 0
    corner(Circle, 7)
    local enabled = default
    local function updateVisuals()
        if enabled then
            tween(Toggle, 0.2, {BackgroundColor3 = Theme.Accent})
            tween(Circle, 0.25, {Position = UDim2.new(1, -17, 0.5, -7)}, Enum.EasingStyle.Back)
            tween(TextLabel, 0.2, {TextColor3 = Theme.TextPrimary})
        else
            tween(Toggle, 0.2, {BackgroundColor3 = Theme.Border})
            tween(Circle, 0.25, {Position = UDim2.new(0, 3, 0.5, -7)}, Enum.EasingStyle.Back)
            tween(TextLabel, 0.2, {TextColor3 = Theme.TextSecondary})
        end
    end
    updateVisuals()
    Holder.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            enabled = not enabled
            updateVisuals()
            callback(enabled)
        end
        end)
    Holder.MouseEnter:Connect(function()
        tween(hStroke, 0.15, {Transparency = 0.3})
        tween(Holder, 0.15, {BackgroundTransparency = 0.2})
    end)
    Holder.MouseLeave:Connect(function()
        tween(hStroke, 0.15, {Transparency = 0.6})
        tween(Holder, 0.15, {BackgroundTransparency = 0.4})
    end)
end

local function CreateSlider(parent, text, layoutOrder, min, max, default, callback, step)
    step = step or 1
    local Holder = Instance.new("Frame", parent)
    Holder.Size = UDim2.new(1, 0, 0, 54)
    Holder.BackgroundColor3 = Theme.SurfaceAlt
    Holder.BackgroundTransparency = 0.4
    Holder.BorderSizePixel = 0
    Holder.LayoutOrder = layoutOrder
    corner(Holder, 10)
    local hStroke = stroke(Holder, Theme.Border, 1, 0.6)
    local Label = Instance.new("TextLabel", Holder)
    Label.BackgroundTransparency = 1
    Label.Position = UDim2.new(0, 14, 0, 8)
    Label.Size = UDim2.new(1, -80, 0, 16)
    Label.Text = text
    Label.Font = Enum.Font.GothamMedium
    Label.TextSize = 12
    Label.TextColor3 = Theme.TextSecondary
    Label.TextXAlignment = Enum.TextXAlignment.Left
    local ValuePill = Instance.new("Frame", Holder)
    ValuePill.AnchorPoint = Vector2.new(1, 0)
    ValuePill.Position = UDim2.new(1, -12, 0, 8)
    ValuePill.Size = UDim2.new(0, 56, 0, 18)
    ValuePill.BackgroundColor3 = Theme.Background
    ValuePill.BorderSizePixel = 0
    corner(ValuePill, 9)
    stroke(ValuePill, Theme.Border, 1, 0.5)
    local ValueLabel = Instance.new("TextLabel", ValuePill)
    ValueLabel.BackgroundTransparency = 1
    ValueLabel.Size = UDim2.new(1, 0, 1, 0)
    ValueLabel.Font = Enum.Font.GothamMedium
    ValueLabel.TextSize = 10
    ValueLabel.TextColor3 = Theme.AccentSoft
    ValueLabel.Text = (step >= 1) and tostring(math.floor(default)) or string.format("%.1f", default)
    local Track = Instance.new("Frame", Holder)
    Track.Position = UDim2.new(0, 14, 1, -18)
    Track.Size = UDim2.new(1, -28, 0, 5)
    Track.BackgroundColor3 = Theme.Border
    Track.BorderSizePixel = 0
    corner(Track, 3)
    local Fill = Instance.new("Frame", Track)
    Fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    Fill.BackgroundColor3 = Theme.Accent
    Fill.BorderSizePixel = 0
    corner(Fill, 3)
    local FillGradient = Instance.new("UIGradient", Fill)
    FillGradient.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, Theme.AccentDark),
        ColorSequenceKeypoint.new(1, Theme.Accent),
    }
    local Knob = Instance.new("Frame", Track)
    Knob.AnchorPoint = Vector2.new(0.5, 0.5)
    Knob.Size = UDim2.new(0, 12, 0, 12)
    Knob.Position = UDim2.new((default - min) / (max - min), 0, 0.5, 0)
    Knob.BackgroundColor3 = Color3.new(1, 1, 1)
    Knob.BorderSizePixel = 0
    corner(Knob, 6)
    stroke(Knob, Theme.Accent, 2, 0)
    local function setValue(val)
        val = math.clamp(val, min, max)
        if step >= 1 then val = math.floor(val + 0.5) end
        local frac = (val - min) / (max - min)
        Fill.Size = UDim2.new(frac, 0, 1, 0)
        Knob.Position = UDim2.new(frac, 0, 0.5, 0)
        ValueLabel.Text = (step >= 1) and tostring(val) or string.format("%.1f", val)
        callback(val)
    end
    local dragging = false
    local function onInput(input)
        local relX = math.clamp(input.Position.X - Track.AbsolutePosition.X, 0, Track.AbsoluteSize.X)
        setValue(min + relX / Track.AbsoluteSize.X * (max - min))
    end
    Holder.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            if input.Position.Y >= Track.AbsolutePosition.Y - 10 then
                dragging = true
                tween(Knob, 0.15, {Size = UDim2.new(0, 16, 0, 16)})
                onInput(input)
            end
        end
    end)
    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
            onInput(input)
        end
    end)
    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            if dragging then tween(Knob, 0.15, {Size = UDim2.new(0, 12, 0, 12)}) end
            dragging = false
        end
    end)
    Holder.MouseEnter:Connect(function() tween(hStroke, 0.15, {Transparency = 0.3}) end)
    Holder.MouseLeave:Connect(function() tween(hStroke, 0.15, {Transparency = 0.6}) end)
end

local function SwitchTab(name)
    for _, v in pairs(Pages) do v.Visible = false end
    Pages[name].Visible = true
    activeTab = name

local lastHitResult = true
local currentResolverInvert = false
local velocityHistory = {}

local function smoothVelocity(userId, currentVel)
    if not velocityHistory[userId] then velocityHistory[userId] = {} end
    local hist = velocityHistory[userId]
    table.insert(hist, currentVel)
    if #hist > 5 then table.remove(hist, 1) end
    local avg = Vector3.zero
    for _, v in ipairs(hist) do avg = avg + v end
    return avg / #hist
end

local function getBodyPart(character, partName)
    if partName == "Head" then return character:FindFirstChild("Head")
    elseif partName == "Body" then return character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Torso")
    elseif partName == "Arms" then return character:FindFirstChild("LeftUpperArm") or character:FindFirstChild("RightUpperArm")
    elseif partName == "Legs" then return character:FindFirstChild("LeftUpperLeg") or character:FindFirstChild("RightUpperLeg")
    else return character:FindFirstChild("HumanoidRootPart")
    end
end

local function getExtremeMultipoints(character, partName)
    local part = getBodyPart(character, partName)
    if not part then
        local fallback = character:FindFirstChild("HumanoidRootPart")
        if fallback then return {fallback.Position} end
        return {}
    end
    local points = {}
    local size = part.Size * 0.6
    local cf = part.CFrame
    table.insert(points, cf.Position)
    table.insert(points, (cf * CFrame.new(size.X/2, 0, 0)).Position)
    table.insert(points, (cf * CFrame.new(-size.X/2, 0, 0)).Position)
    table.insert(points, (cf * CFrame.new(0, size.Y/2, 0)).Position)
    table.insert(points, (cf * CFrame.new(0, -size.Y/2, 0)).Position)
    return points
end

local function getBestTarget()
    local camPos = Workspace.CurrentCamera.CFrame.Position
    local camLook = Workspace.CurrentCamera.CFrame.LookVector
    local bestTarget, bestDist = nil, Settings.MaxDistance
    for _, other in ipairs(Players:GetPlayers()) do
        if other == player or (player.Team and other.Team and player.Team == other.Team) then continue end
        local chr = other.Character
        if not chr or not chr:FindFirstChild("Humanoid") or chr.Humanoid.Health <= 0 then continue end
        local points = getExtremeMultipoints(chr, Settings.HitboxMode)
        if #points == 0 then continue end
        local chosenPoint = points[1]
        local foundVis = false
        for _, pt in ipairs(points) do
            local ray = Workspace:Raycast(camPos, pt - camPos)
            if not ray or ray.Instance:IsDescendantOf(chr) then
                chosenPoint = pt
                foundVis = true
                break
            end
        end
        if not foundVis and not Settings.Wallbang then continue end
        local dist = (chosenPoint - camPos).Magnitude
        if dist > bestDist then continue end
        if Settings.FOV < 360 then
            local dir = (chosenPoint - camPos).Unit
            local angle = math.acos(math.clamp(camLook:Dot(dir), -1, 1))
            if math.deg(angle) > Settings.FOV then continue end
        end
        bestTarget = {
            player = other, character = chr, position = chosenPoint,
            part = getBodyPart(chr, Settings.HitboxMode) or chr.HumanoidRootPart
        }
        bestDist = dist
    end
    return bestTarget
end

local function getAdaptivePrediction(target, finalPos)
    local currentVel = target.part and target.part.AssemblyLinearVelocity or Vector3.zero
    local userId = target.player.UserId
    local smoothVel = smoothVelocity(userId, currentVel)
    local hist = velocityHistory[userId]
    local acceleration = Vector3.zero
    if hist and #hist >= 2 then
        local prevVel = hist[#hist-1]
        acceleration = (currentVel - prevVel) / 0.016
    end
    local ping = Stats.Network.ServerStatsItem["Data Ping"]:GetValue() / 1000
    local dist = (finalPos - Workspace.CurrentCamera.CFrame.Position).Magnitude
    local bulletTime = (Settings.BulletSpeed < 9999) and (dist / Settings.BulletSpeed) or 0.03
    local totalTime = bulletTime + ping + 0.05
    if Settings.AdaptivePredict then
        return finalPos + smoothVel * (totalTime * Settings.PredictionStrength) + 0.5 * acceleration * totalTime * totalTime
    else
        return finalPos + smoothVel * (totalTime * Settings.PredictionStrength)
    end
end

local function calculateResolver(target)
    if Settings.ResolverMode == "Off" then return Vector3.zero end
    if not lastHitResult then
        currentResolverInvert = not currentResolverInvert
        lastHitResult = true
    end
    local multiplier = currentResolverInvert and 1 or -1
    local currentVel = target.part and target.part.AssemblyLinearVelocity or Vector3.zero
    local smoothVel = smoothVelocity(target.player.UserId, currentVel)
    local perp = Vector3.new(-smoothVel.Z, 0, smoothVel.X).Unit
    return perp * Settings.ResolverStrength * multiplier
end

local jitterFlip = false
local lastFlipTime = 0
local lagCounter = 0
local cachedCFrame = nil
local savedAutoRotate = nil
local realCF = nil

local function shouldRunAA(hum, hrp)
    if not Settings.AAEnabled then return false end
    if not hrp or not hum or hum.Health <= 0 then return false end
    local moving = hum.MoveDirection.Magnitude > 0.1
    local inAir = hum.FloorMaterial == Enum.Material.Air
    if moving and not Settings.AAOnMove then return false end
    if inAir and not Settings.AAOnAir then return false end
    return true
end

local function computePitch()
    local m = Settings.AAPitch
    if m == "Down"   then return math.rad(89)
    elseif m == "Up"     then return math.rad(-89)
    elseif m == "Zero"   then return 0
    elseif m == "Custom" then return math.rad(Settings.AAPitchCustom)
    end
    return 0
end

local function computeBaseYaw()
    local cam = Workspace.CurrentCamera
    local lv = cam.CFrame.LookVector
    local base = math.atan2(-lv.X, -lv.Z)
    local m = Settings.AAYawBase
    if m == "Backward" then base = base + math.pi
    elseif m == "Left" then base = base + math.pi/2
    elseif m == "Right" then base = base - math.pi/2
    end
    return base
end

local function computeYaw()
    local yaw = computeBaseYaw() + math.rad(Settings.AAYawOffset)
    local mod = Settings.AAYawMod
    local now = tick()

    if mod == "Jitter" then
        if now - lastFlipTime > 0.08 then
            jitterFlip = not jitterFlip
            lastFlipTime = now
        end
        yaw = yaw + math.rad(jitterFlip and Settings.AAJitterRange or -Settings.AAJitterRange)
    elseif mod == "Spin" then
        yaw = yaw + now * Settings.AASpinSpeed
    elseif mod == "Random" then
        yaw = yaw + math.rad(math.random(-Settings.AAJitterRange, Settings.AAJitterRange))
    end

    if Settings.AADesync > 0 then
        yaw = yaw + math.sin(now * 18) * math.rad(Settings.AADesync) * 0.5
    end

    if Settings.AAEdgeYaw then
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            local dir = Vector3.new(-math.sin(yaw), 0, -math.cos(yaw)) * 3
            local ray = Workspace:Raycast(hrp.Position, dir)
            if ray then yaw = computeBaseYaw() end
        end
    end

    return yaw
end

-- Применяем fake-угол перед репликацией, сразу восстанавливаем
-- Это не даёт физике уронить персонажа
RunService.Stepped:Connect(function()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")

    if not shouldRunAA(hum, hrp) then
        -- Восстанавливаем AutoRotate, если выключили AA
        if savedAutoRotate ~= nil and hum then
            hum.AutoRotate = savedAutoRotate
            savedAutoRotate = nil
        end
        return
    end

    -- Запоминаем оригинальный AutoRotate и выключаем
    if savedAutoRotate == nil then
        savedAutoRotate = hum.AutoRotate
        hum.AutoRotate = false
    end

    -- Fake Lag: пропуск тиков
    if Settings.AAFakeLag then
        lagCounter = lagCounter + 1
        if lagCounter < Settings.AAFakeLagAmount then
            if cachedCFrame then
                pcall(function() hrp.CFrame = cachedCFrame end)
            end
            return
        else
            lagCounter = 0
        end
    end

    -- Сохраняем настоящий CFrame
    realCF = hrp.CFrame

    local pos = realCF.Position
    local pitch = computePitch()
    local yaw = computeYaw()

    -- Применяем yaw в upright-режиме (Y-up) — не валит
    local upright = CFrame.lookAt(pos, pos + Vector3.new(-math.sin(yaw), 0, -math.cos(yaw)))
    -- Pitch добавляем поверх — но это лишь на один кадр (Heartbeat вернёт обратно)
    local fakeCF = upright * CFrame.Angles(pitch, 0, 0)

    cachedCFrame = fakeCF
    pcall(function() hrp.CFrame = fakeCF end)
end)

    local progress = Instance.new("Frame", toast)
    progress.Position = UDim2.new(0, 0, 1, -2)
    progress.Size = UDim2.new(1, 0, 0, 2)
    progress.BackgroundColor3 = Theme.Accent

                if c:IsA("TextLabel") then tween(c, 0.3, {TextTransparency = 1}) end

                if c:IsA("ImageLabel") then tween(c, 0.3, {ImageTransparency = 1}) end
                if c:IsA("UIStroke") then tween(c, 0.3, {Transparency = 1}) end
            end)
        end
        task.wait(0.35)
        toast:Destroy()
    end)
end

-- ===================== SOUND FX =====================
local function playKillSound()
    if not Settings.KillMsgSound then return end
    pcall(function()
        local s = Instance.new("Sound", gui)
        s.SoundId = "rbxassetid://3120208166" -- short ding
        s.Volume = 0.6
        s.PlayOnRemove = false
        s:Play()
        task.delay(2, function() s:Destroy() end)
    end)
end

-- ===================== CHAT =====================
local function sendChat(msg)
    if not Settings.KillMsgChat then return end
    pcall(function()
        local ev = game:GetService("ReplicatedStorage"):FindFirstChild("DefaultChatSystemChatEvents")
        if ev and ev:FindFirstChild("SayMessageRequest") then
            ev.SayMessageRequest:FireServer(msg, "All")
        else
            -- Новый TextChatService
            local TCS = game:GetService("TextChatService")
            local gen = TCS.TextChannels:FindFirstChild("RBXGeneral")
            if gen then gen:SendAsync(msg) end
        end
    end)
end

-- ===================== ОТСЛЕЖИВАНИЕ УБИЙСТВ =====================
local recentlyAttacked = {} -- [Player] = timestamp

-- Трекаем кого мы атаковали
local origHeartbeat = RunService.Heartbeat
-- Используем хук на наш аим
local oldGetBestTarget = getBestTarget
function getBestTarget(...)  -- упс, нельзя так, getBestTarget = local
    return oldGetBestTarget(...)
end
-- Лучше отдельный хук
local origFire = nil
RunService.Heartbeat:Connect(function()
    if not Settings.Enabled then return end
    -- Если есть текущая цель — отмечаем её
    local tgt = getBestTarget()
    if tgt and tgt.player then
        recentlyAttacked[tgt.player] = tick()
    end
end)

local function bindPlayer(p)
    if p == player then return end
    p.CharacterAdded:Connect(function(char)
        local hum = char:WaitForChild("Humanoid", 5)
        if not hum then return end
        hum.Died:Connect(function()
            if not Settings.KillMsgEnabled then return end
            local lastAtk = recentlyAttacked[p]
            -- Считаем "наш килл" если мы атаковали жертву в последние 2 секунды
            if lastAtk and (tick() - lastAtk) < 2 then
                recentlyAttacked[p] = nil
                if Settings.KillMsgToast then showToast(p.Name) end
                playKillSound()
                sendChat(Settings.KillMsgText .. " → " .. p.Name)
            end
        end)
    end)
    -- Сразу подключаем к существующему персонажу
    if p.Character then
        local hum = p.Character:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.Died:Connect(function()
                if not Settings.KillMsgEnabled then return end
                local lastAtk = recentlyAttacked[p]
                if lastAtk and (tick() - lastAtk) < 2 then
                    recentlyAttacked[p] = nil
                    if Settings.KillMsgToast then showToast(p.Name) end
                    playKillSound()
                    sendChat(Settings.KillMsgText .. " → " .. p.Name)
                end
            end)
        end
    end
end

for _, p in ipairs(Players:GetPlayers()) do bindPlayer(p) end
Players.PlayerAdded:Connect(bindPlayer)
Players.PlayerRemoving:Connect(function(p) recentlyAttacked[p] = nil end)
